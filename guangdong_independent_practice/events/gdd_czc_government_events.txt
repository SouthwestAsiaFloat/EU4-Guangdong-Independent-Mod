# One-time new-game/old-save initialization. The historical, unchanged CZC
# oligarchy also migrates; a different player-selected government does not.
country_event = {
    id = gdd_czc_government.1
    title = gdd_czc_government.10.t
    desc = gdd_czc_government.10.d
    picture = ELECTION_REPUBLICAN_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            gdd_czc_is_council_trigger = yes
            gdd_czc_should_migrate_legacy_trigger = yes
        }
    }
    immediate = {
        gdd_czc_migrate_legacy_council_effect = yes
        gdd_czc_initialize_effect = yes
        gdd_czc_check_transition_effect = yes
    }
    option = { name = gdd_czc_government.10.c }
}

# Native modifiers own yearly republican-tradition changes. This event only
# catches the transition and gives the AI access to the same annual interaction.
country_event = {
    id = gdd_czc_government.2
    title = gdd_czc_government.10.t
    desc = gdd_czc_government.10.d
    picture = ELECTION_REPUBLICAN_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            gdd_czc_is_council_trigger = yes
            gdd_czc_should_migrate_legacy_trigger = yes
        }
    }
    immediate = {
        gdd_czc_migrate_legacy_council_effect = yes
        gdd_czc_initialize_effect = yes
        gdd_czc_check_transition_effect = yes
        if = {
            limit = {
                ai = yes
                gdd_czc_can_shift_power_trigger = yes
            }
            random_list = {
                50 = { gdd_czc_support_gentry_effect = yes }
                50 = { gdd_czc_support_ruler_effect = yes }
            }
        }
    }
    option = { name = gdd_czc_government.10.c }
}

# Console-only diagnostics; never scheduled by ordinary gameplay.
# event gdd_czc_government.900 CZC : install in an isolated test campaign.
country_event = {
    id = gdd_czc_government.900
    title = gdd_czc_government.10.t
    desc = gdd_czc_government.10.d
    picture = ELECTION_REPUBLICAN_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = { tag = CZC }
    immediate = {
        clr_country_flag = gdd_czc_crowned
        clr_country_flag = gdd_czc_council_initialized
        clr_country_flag = gdd_czc_last_council_action
        change_government = republic
        remove_government_reform = oligarchy_reform
        add_government_reform = gdd_czc_council_reform
        gdd_czc_initialize_effect = yes
        # Explicit test reset, including tests loaded after the 1444 start date.
        add_republican_tradition = -100
        add_republican_tradition = 50
    }
    option = { name = gdd_czc_government.10.c }
}
