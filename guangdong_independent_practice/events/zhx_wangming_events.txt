# Registered only on the native monthly pulse, which supplies its own game seed.
# A GUI-triggered country_event (even delayed) would inherit the fixed GUI seed.
country_event = {
    id = zhx_wangming.1
    hidden = yes
    is_triggered_only = yes
    trigger = {
        zhx_tianzi_has_government = yes
        has_global_flag = zhx_system_initialised_v13
        OR = {
            has_global_flag = zhx_wm_tribute_queued
            has_global_flag = zhx_wm_migration_queued
            has_global_flag = zhx_wm_levy_queued
        }
    }
    immediate = {
        if = {
            limit = { has_global_flag = zhx_wm_tribute_queued }
            clr_global_flag = zhx_wm_tribute_queued
            zhx_wm_issue_tribute = yes
        }
        if = {
            limit = { has_global_flag = zhx_wm_migration_queued }
            clr_global_flag = zhx_wm_migration_queued
            zhx_wm_issue_migration = yes
        }
        if = {
            limit = { has_global_flag = zhx_wm_levy_queued }
            clr_global_flag = zhx_wm_levy_queued
            zhx_wm_issue_levy = yes
        }
    }
}
