# If Jiyuan succeeds Shouqian, dies first, or is disinherited first, Chengzuo
# becomes the next heir. This event is called before gdd_startup.6 clears the
# Shouqian-reigning flag on succession.
country_event = {
    id = gdd_heir.2
    hidden = yes
    is_triggered_only = yes

    trigger = {
        tag = GDD
        has_country_flag = gdd_liang_shouqian_reigning
        has_country_flag = gdd_liang_shouyi_designated_heir
        NOT = { has_country_flag = gdd_liang_chengzuo_created }
    }

    immediate = {
        clr_country_flag = gdd_liang_shouyi_designated_heir
        set_country_flag = gdd_liang_chengzuo_created
        set_country_flag = gdd_liang_chengzuo_pending

        # Capture eligibility before gdd_startup.6 clears the reigning flag,
        # but let the native death/disinheritance/succession transaction finish.
        # EU4 1.37.5 otherwise clears or replaces a synchronously defined heir.
        country_event = { id = gdd_heir.3 days = 1 }
    }
}
