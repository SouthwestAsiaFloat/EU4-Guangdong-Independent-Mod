# 进入和谈或省份易主事务。
country_event = {
    id = zhx_diplomacy.11
    title = zhx_diplomacy.10.t
    desc = zhx_diplomacy.10.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                NOT = { zhx_is_tianzi = yes }
                has_saved_global_event_target = zhx_tianzi
                event_target:zhx_tianzi = { exists = yes }
            }
            event_target:zhx_tianzi = {
                add_opinion = {
                    who = ROOT
                    modifier = zhx_opinion_covenant_breaker
                }
            }
        }
    }

    option = { name = OK }
}

# 月度犯盟战争理由桥。与关系桥拆开，便于单独验证、回滚和定位问题。
# ROOT 是犯盟责任国；战争理由授予天子，目标保持为稳定的事件 ROOT。
country_event = {
    id = zhx_diplomacy.12
    title = zhx_diplomacy.10.t
    desc = zhx_diplomacy.10.d
    picture = DIPLOMACY_eventPicture
    hidden = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                NOT = { zhx_is_tianzi = yes }
                has_saved_global_event_target = zhx_tianzi
                event_target:zhx_tianzi = { exists = yes }
            }
            event_target:zhx_tianzi = {
                # This mirrors native triggered-event CB grants. Expiry matches
                # the ten-year covenant-breaker state; reoffending refreshes it.
                add_casus_belli = {
                    target = ROOT
                    type = zhx_cb_punish_covenant_breaker
                    months = 120
                }
            }
        }
    }

    option = { name = OK }
}

                factor = 3
                alliance_with = event_target:zhx_diplomacy_applicant

                factor = 2

                }
            }
            modifier = {
                factor = 0
                is_rival = event_target:zhx_diplomacy_applicant

        name = zhx_diplomacy.30.b
        custom_tooltip = zhx_diplomacy_reject_admission_tt

                }
            }
        }

    }
