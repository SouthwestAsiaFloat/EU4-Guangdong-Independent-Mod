# Route reforms unlock after any five of the eight independent ordinary
# reforms. Centralising and decentralising reforms never count toward this
# threshold.
gdd_five_ordinary_celestial_reforms_passed = {
    calc_true_if = {
        gdd_reform_civil_registration_passed = yes
        gdd_reform_silver_standard_passed = yes
        gdd_reform_kanhe_passed = yes
        gdd_reform_keju_passed = yes
        gdd_reform_inclusive_monarchy_passed = yes
        gdd_reform_unified_market_passed = yes
        gdd_reform_military_branch_passed = yes
        gdd_reform_foreign_ship_designs_passed = yes
        amount = 5
    }
}
