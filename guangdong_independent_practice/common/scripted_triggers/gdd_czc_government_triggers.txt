gdd_czc_is_council_trigger = {
    tag = CZC
    government = republic
    has_reform = gdd_czc_council_reform
    NOT = { has_country_flag = gdd_czc_crowned }
}

gdd_czc_should_migrate_legacy_trigger = {
    tag = CZC
    government = republic
    has_reform = oligarchy_reform
    NOT = { has_country_flag = gdd_czc_crowned }
    NOT = { has_country_flag = gdd_czc_council_initialized }
}

gdd_czc_can_shift_power_trigger = {
    gdd_czc_is_council_trigger = yes
    adm_power = 50
    OR = {
        NOT = { has_country_flag = gdd_czc_last_council_action }
        had_country_flag = { flag = gdd_czc_last_council_action days = 365 }
    }
}

gdd_czc_should_proclaim_king_trigger = {
    gdd_czc_is_council_trigger = yes
    NOT = { republican_tradition = 20 }
}
