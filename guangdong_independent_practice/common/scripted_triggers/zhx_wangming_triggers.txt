# Country scopes; the authority is the existing global Tianzi target.
zhx_wm_member = {
    exists = yes
    zhx_is_tianxia_member = yes
    NOT = { zhx_is_tianzi = yes }
    has_saved_global_event_target = zhx_tianzi
    NOT = { war_with = event_target:zhx_tianzi }
}
zhx_wm_can_tribute = {
    zhx_tianzi_has_government = yes
    has_global_flag = zhx_system_initialised_v13
    NOT = { has_global_flag = zhx_wm_tribute_used }
    NOT = { has_global_flag = zhx_wm_tribute_queued }
    any_country = { zhx_wm_member = yes }
}
zhx_wm_can_pay = {
    variable_arithmetic_trigger = {
        export_to_variable = { variable_name = zhx_wm_cash value = treasury }
        check_variable = { which = zhx_wm_cash which = zhx_wm_tribute_due }
    }
}
zhx_wm_tribute_valid = {
    has_country_flag = zhx_wm_tribute_pending
    zhx_wm_member = yes
    has_saved_event_target = zhx_wm_sender
    event_target:zhx_wm_sender = { exists = yes zhx_tianzi_has_government = yes }
    NOT = { war_with = event_target:zhx_wm_sender }
}
