# Country scope; includes human and AI owners without a viewer/ROOT-country filter.
# Also clears former members. Monthly repair handles engine modifier removal.
zhx_refresh_country_ritual_warscore = {
    every_owned_province = {
        limit = { OR = { owner = { zhx_is_tianxia_polity = yes } has_province_flag = zhx_ritual_warscore_active } }
        zhx_refresh_province_ritual_warscore = yes
    }
}

# Called by the existing authority refresh, independently of its five country tiers.
zhx_refresh_all_ritual_warscore = {
    every_country = {
        limit = { exists = yes }
        zhx_refresh_country_ritual_warscore = yes
    }
}
