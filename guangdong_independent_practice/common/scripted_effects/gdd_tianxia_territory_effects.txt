# Province scope. Adds one eligible province and awards its one-time Mandate
# grant: 0.1 per development, capped at 20 for this province.
gdd_add_province_to_tianxia_effect = {
    if = {
        limit = {
            NOT = { has_global_flag = zhx_tianxia_dismantled }
            NOT = { has_province_flag = zhx_tianxia_province }
            is_part_of_hre = no
            is_owned_by_trade_company = no
            owner = { zhx_is_tianxia_polity = yes }
        }
        set_province_flag = zhx_tianxia_province
        if = {
            limit = {
                NOT = { has_province_flag = zhx_tianxia_admission_reward_paid }
                any_country = { is_emperor_of_china = yes }
            }
            set_province_flag = zhx_tianxia_admission_reward_paid
            owner = {
                export_to_variable = {
                    which = gdd_tianxia_province_mandate_amount
                    value = development
                    who = PREV
                }
                if = {
                    limit = {
                        check_variable = {
                            which = gdd_tianxia_province_mandate_amount
                            value = 200
                        }
                    }
                    set_variable = {
                        which = gdd_tianxia_province_mandate_amount
                        value = 200
                    }
                }
                while = {
                    limit = {
                        check_variable = {
                            which = gdd_tianxia_province_mandate_amount
                            value = 1
                        }
                    }
                    every_country = {
                        limit = { is_emperor_of_china = yes }
                        add_mandate = 0.1
                    }
                    change_variable = {
                        which = gdd_tianxia_province_mandate_amount
                        value = -1
                    }
                }
                set_variable = {
                    which = gdd_tianxia_province_mandate_amount
                    value = 0
                }
            }
        }
    }
}

# Province scope. Removes one province and charges 0.1 Mandate per development
# (capped at 20 for this province). Opinion is applied from country scope by
# the caller so the current Emperor always receives the correct opinion target.
# The permanent reward-paid flag is intentionally retained.
gdd_remove_province_from_tianxia_effect = {
    if = {
        limit = {
            has_province_flag = zhx_tianxia_province
            owner = {
                exists = yes
                NOT = { zhx_is_tianxia_polity = yes }
            }
            any_country = { is_emperor_of_china = yes }
        }
        owner = {
            export_to_variable = {
                which = gdd_tianxia_province_mandate_amount
                value = development
                who = PREV
            }
            if = {
                limit = {
                    check_variable = {
                        which = gdd_tianxia_province_mandate_amount
                        value = 200
                    }
                }
                set_variable = {
                    which = gdd_tianxia_province_mandate_amount
                    value = 200
                }
            }
            while = {
                limit = {
                    check_variable = {
                        which = gdd_tianxia_province_mandate_amount
                        value = 1
                    }
                }
                every_country = {
                    limit = { is_emperor_of_china = yes }
                    add_mandate = -0.1
                }
                change_variable = {
                    which = gdd_tianxia_province_mandate_amount
                    value = -1
                }
            }
            set_variable = {
                which = gdd_tianxia_province_mandate_amount
                value = 0
            }
        }
        clr_province_flag = zhx_tianxia_province
    }
}

# Country scope: the AI country responsible for removing the province. Preserve
# that current country only for the duration of this same scripted effect, then
# make the Emperor add the opinion from the correct direction. Human removals
# use a still more explicit country-event path in gdd_tianxia_territory.2.
gdd_apply_tianxia_removal_opinion_effect = {
    save_event_target_as = gdd_tianxia_removal_offender
    if = {
        limit = {
            event_target:EmperorOfChina = {
                exists = yes
                is_emperor_of_china = yes
            }
        }
        event_target:EmperorOfChina = {
            add_opinion = {
                who = event_target:gdd_tianxia_removal_offender
                modifier = gdd_opinion_removed_tianxia_province
                years = 100
            }
        }
    }
    else_if = {
        limit = { CZH = { exists = yes is_emperor_of_china = yes } }
        CZH = {
            add_opinion = {
                who = event_target:gdd_tianxia_removal_offender
                modifier = gdd_opinion_removed_tianxia_province
                years = 100
            }
        }
    }
}

# Country scope: consider this holder's eligible provinces. ROOT remains the
# Emperor and owns the two comparison variables. Equal-development ties follow
# the engine's stable province iteration.
gdd_consider_unlawful_tianxia_provinces_for_holder_effect = {
    every_owned_province = {
        limit = {
            gdd_is_unlawful_tianxia_province_for_country_trigger = {
                who = PREV
            }
        }
        ROOT = {
            export_to_variable = {
                which = gdd_tianxia_unlawful_candidate_development
                value = development
                who = PREV
            }
            if = {
                limit = {
                    check_variable = {
                        which = gdd_tianxia_unlawful_candidate_development
                        which = gdd_tianxia_unlawful_best_development
                    }
                }
                set_variable = {
                    which = gdd_tianxia_unlawful_best_development
                    which = gdd_tianxia_unlawful_candidate_development
                }
                set_country_flag = gdd_tianxia_unlawful_province_selected
                PREV = {
                    save_event_target_as = gdd_tianxia_unlawful_province
                    owner = {
                        save_event_target_as = gdd_tianxia_unlawful_actual_owner
                    }
                }
            }
        }
    }
}

# ROOT is the Emperor and $target$ is the sovereign diplomacy recipient. Search
# the recipient first and then its direct subjects, retaining one deterministic
# highest-development province across the whole sovereignty group.
gdd_select_highest_development_unlawful_tianxia_province_for_addressee_effect = {
    clr_country_flag = gdd_tianxia_unlawful_province_selected
    set_variable = {
        which = gdd_tianxia_unlawful_best_development
        value = -1
    }
    set_variable = {
        which = gdd_tianxia_unlawful_candidate_development
        value = 0
    }
    $target$ = {
        if = {
            limit = {
                gdd_can_receive_tianxia_unlawful_demand_trigger = {
                    emperor = ROOT
                }
            }
            gdd_consider_unlawful_tianxia_provinces_for_holder_effect = yes
        }
        every_subject_country = {
            limit = {
                gdd_can_receive_tianxia_unlawful_demand_trigger = {
                    emperor = ROOT
                }
            }
            gdd_consider_unlawful_tianxia_provinces_for_holder_effect = yes
        }
    }
    set_variable = {
        which = gdd_tianxia_unlawful_best_development
        value = 0
    }
    set_variable = {
        which = gdd_tianxia_unlawful_candidate_development
        value = 0
    }
}

# ROOT is the human Emperor and FROM is the native diplomatic recipient. Native
# requests cannot store a province at send-time, so resolution re-selects and
# revalidates the highest-development eligible province before applying either
# outcome.
gdd_prepare_native_tianxia_unlawful_context_effect = {
    save_event_target_as = gdd_tianxia_unlawful_emperor
    FROM = { save_event_target_as = gdd_tianxia_unlawful_addressee }
    gdd_select_highest_development_unlawful_tianxia_province_for_addressee_effect = {
        target = FROM
    }
}

# Country scope: calculate the recipient AI's deterministic acceptance score.
# Component flags exist only for the open confirmation/request event and drive
# its readable reasons; the cleanup effect removes every scratch value.
gdd_prepare_tianxia_unlawful_acceptance_effect = {
    gdd_clear_tianxia_unlawful_acceptance_cache_effect = yes
    set_variable = {
        which = gdd_tianxia_unlawful_acceptance_score
        value = 10
    }

    if = {
        limit = {
            event_target:gdd_tianxia_unlawful_actual_owner = {
                is_subject_of = event_target:gdd_tianxia_unlawful_emperor
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 100
        }
        set_country_flag = gdd_tianxia_unlawful_reason_emperor_subject
    }

    if = {
        limit = {
            event_target:gdd_tianxia_unlawful_emperor = {
                army_strength = { who = ROOT value = 1.5 }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 40
        }
        set_country_flag = gdd_tianxia_unlawful_reason_emperor_much_stronger
    }
    else_if = {
        limit = {
            event_target:gdd_tianxia_unlawful_emperor = {
                army_strength = { who = ROOT value = 1.1 }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 20
        }
        set_country_flag = gdd_tianxia_unlawful_reason_emperor_stronger
    }
    else_if = {
        limit = {
            army_strength = {
                who = event_target:gdd_tianxia_unlawful_emperor
                value = 1.5
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -40
        }
        set_country_flag = gdd_tianxia_unlawful_reason_recipient_stronger
    }

    if = {
        limit = {
            has_opinion = {
                who = event_target:gdd_tianxia_unlawful_emperor
                value = 100
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 20
        }
        set_country_flag = gdd_tianxia_unlawful_reason_opinion_very_high
    }
    else_if = {
        limit = {
            has_opinion = {
                who = event_target:gdd_tianxia_unlawful_emperor
                value = 50
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 10
        }
        set_country_flag = gdd_tianxia_unlawful_reason_opinion_high
    }
    else_if = {
        limit = {
            NOT = {
                has_opinion = {
                    who = event_target:gdd_tianxia_unlawful_emperor
                    value = -50
                }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -25
        }
        set_country_flag = gdd_tianxia_unlawful_reason_opinion_very_low
    }
    else_if = {
        limit = {
            NOT = {
                has_opinion = {
                    who = event_target:gdd_tianxia_unlawful_emperor
                    value = 0
                }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -15
        }
        set_country_flag = gdd_tianxia_unlawful_reason_opinion_low
    }

    if = {
        limit = {
            trust = {
                who = event_target:gdd_tianxia_unlawful_emperor
                value = 80
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 20
        }
        set_country_flag = gdd_tianxia_unlawful_reason_trust_very_high
    }
    else_if = {
        limit = {
            trust = {
                who = event_target:gdd_tianxia_unlawful_emperor
                value = 60
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 10
        }
        set_country_flag = gdd_tianxia_unlawful_reason_trust_high
    }
    else_if = {
        limit = {
            NOT = {
                trust = {
                    who = event_target:gdd_tianxia_unlawful_emperor
                    value = 40
                }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -10
        }
        set_country_flag = gdd_tianxia_unlawful_reason_trust_low
    }

    if = {
        limit = { alliance_with = event_target:gdd_tianxia_unlawful_emperor }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = 15
        }
        set_country_flag = gdd_tianxia_unlawful_reason_allied
    }
    if = {
        limit = {
            OR = {
                is_rival = event_target:gdd_tianxia_unlawful_emperor

                    who = event_target:gdd_tianxia_unlawful_emperor
                    attitude = attitude_hostile

        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -35
        }
        set_country_flag = gdd_tianxia_unlawful_reason_hostile
    }

    if = {

            event_target:gdd_tianxia_unlawful_emperor = {
                diplomatic_reputation = 5

        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score

            which = gdd_tianxia_unlawful_acceptance_score

        set_country_flag = gdd_tianxia_unlawful_reason_diprep_high
    }

            event_target:gdd_tianxia_unlawful_emperor = {
                NOT = { diplomatic_reputation = 0 }
            }
        }
        change_variable = {
            which = gdd_tianxia_unlawful_acceptance_score
            value = -10
        }
        set_country_flag = gdd_tianxia_unlawful_reason_diprep_low
    }

    if = {
        limit = {
            check_variable = {
                which = gdd_tianxia_unlawful_acceptance_score
                value = 0
            }
        }
        set_country_flag = gdd_tianxia_unlawful_ai_will_accept
    }
}

# ROOT is the Emperor. The local event targets were prepared by the
# confirmation event and travel with the request to its addressee.
gdd_dispatch_tianxia_unlawful_demand_effect = {
    event_target:gdd_tianxia_unlawful_province = {
        set_province_flag = gdd_tianxia_unlawful_request_pending
    }
    event_target:gdd_tianxia_unlawful_actual_owner = {
        set_country_flag = gdd_tianxia_unlawful_request_pending
        clr_country_flag = gdd_tianxia_unlawful_demand_cooldown
        set_country_flag = gdd_tianxia_unlawful_demand_cooldown
    }
    event_target:gdd_tianxia_unlawful_addressee = {
        set_country_flag = gdd_tianxia_unlawful_request_pending
    }
    if = {
        # When the unlawful holder is the Emperor's own subject, sovereignty
        # makes the Emperor the addressee. The confirmed demand is therefore
        # enforced directly instead of sending an event back to its sender.
        limit = {
            event_target:gdd_tianxia_unlawful_addressee = {
                is_emperor_of_china = yes
            }
        }
        gdd_accept_tianxia_unlawful_demand_effect = yes
    }
    else = {
        event_target:gdd_tianxia_unlawful_addressee = {
            country_event = { id = gdd_tianxia_territory.11 }
        }
    }
}

# Country scope within the province's core-country iterator. A marker on the

# Re-resolve against live cores immediately before restitution. Core-country
# iteration includes extinct tags, unlike every_country. Priority: direct
# previous lawful owner, other former lawful owners, then other lawful cores.
gdd_select_lawful_tianxia_returnee_effect = {
    event_target:gdd_tianxia_unlawful_emperor = {
        clr_country_flag = gdd_tianxia_unlawful_returnee_selected
    }
    event_target:gdd_tianxia_unlawful_province = {
        every_core_country = {
            limit = {
                gdd_is_lawful_tianxia_returnee_trigger = yes
                event_target:gdd_tianxia_unlawful_province = { previous_owner = PREV }
                event_target:gdd_tianxia_unlawful_emperor = {
                    NOT = { has_country_flag = gdd_tianxia_unlawful_returnee_selected }
                }
            }
            gdd_select_tianxia_returnee_candidate_effect = yes
        }
        every_core_country = {
            limit = {
                gdd_is_lawful_tianxia_returnee_trigger = yes
                has_country_flag = gdd_tianxia_former_legal_owner
                event_target:gdd_tianxia_unlawful_emperor = {
                    NOT = { has_country_flag = gdd_tianxia_unlawful_returnee_selected }
                }
            }
            gdd_select_tianxia_returnee_candidate_effect = yes
        }
        every_core_country = {
            limit = {
                gdd_is_lawful_tianxia_returnee_trigger = yes
                event_target:gdd_tianxia_unlawful_emperor = {
                    NOT = { has_country_flag = gdd_tianxia_unlawful_returnee_selected }
                }
            }
            gdd_select_tianxia_returnee_candidate_effect = yes
        }
    }
}

# Country scope; shared by native diplomacy and the AI event path. Every
# material mutation is guarded by the same core/ownership eligibility used by
# the button, so stale requests cannot punish or transfer a now-lawful province.
gdd_accept_tianxia_unlawful_demand_effect = {
    if = {
        limit = {
            event_target:gdd_tianxia_unlawful_province = {
                gdd_is_substantively_unlawful_tianxia_province_trigger = {
                    who = event_target:gdd_tianxia_unlawful_actual_owner
                }
            }
        }
        gdd_select_lawful_tianxia_returnee_effect = yes
        if = {
            limit = {
                event_target:gdd_tianxia_unlawful_emperor = {
                    has_country_flag = gdd_tianxia_unlawful_returnee_selected
                }
            }
            event_target:gdd_tianxia_unlawful_actual_owner = {
                add_opinion = {
                    who = event_target:gdd_tianxia_unlawful_emperor
                    modifier = gdd_opinion_forced_tianxia_return
                    years = 50
                }
            }
            event_target:gdd_tianxia_unlawful_province = {
                remove_claim = event_target:gdd_tianxia_unlawful_actual_owner
                remove_province_modifier = gdd_tianxia_unlawful_refusal
                # Cede exactly this province; release would transfer all of
                # the extinct tag's cores owned by the current holder.
                cede_province = event_target:gdd_tianxia_unlawful_returnee
            }
            event_target:gdd_tianxia_unlawful_returnee = {
                # Registration restores only ordinary membership. It must not
                # steal the Tianzi office or a great-feudatory seat.
                if = {
                    limit = {
                        exists = yes
                        NOT = { zhx_is_tianxia_polity = yes }
                    }
                    zhx_register_tianxia_member = yes
                    country_event = { id = zhx_doctrine.92 }
                }
            }

# Country scope. Voluntary departure removes all directly owned Tianxia
# provinces, but charges a single aggregate development loss capped at 20.
gdd_leave_tianxia_territory_effect = {
    set_variable = {
        which = gdd_tianxia_exit_development
        value = 0
    }
    every_owned_province = {
        limit = { has_province_flag = zhx_tianxia_province }
        PREV = {
            export_to_variable = {
                which = gdd_tianxia_exit_province_development
                value = development
                who = PREV
            }
            change_variable = {
                which = gdd_tianxia_exit_development
                which = gdd_tianxia_exit_province_development
            }
            set_variable = {
                which = gdd_tianxia_exit_province_development
                value = 0
            }
        }
        clr_province_flag = zhx_tianxia_province
    }
    if = {
        limit = {
            check_variable = {
                which = gdd_tianxia_exit_development
                value = 200
            }
        }
        set_variable = {
            which = gdd_tianxia_exit_development
            value = 200
        }
    }
    while = {
        limit = {
            check_variable = {
                which = gdd_tianxia_exit_development
                value = 1
            }
        }
        event_target:EmperorOfChina = { add_mandate = -0.1 }
        change_variable = {
            which = gdd_tianxia_exit_development
            value = -1
        }
    }
    set_variable = {
        which = gdd_tianxia_exit_development
        value = 0
    }
    event_target:EmperorOfChina = {
        add_opinion = {
            who = ROOT
            modifier = gdd_opinion_left_tianxia
            years = 50
        }
    }
}

# Quarterly AI maintenance and mutual-exclusion repair. The AI Emperor considers
# at most one unlawful-territory demand per quarterly tick; independent holder
# cooldowns and pending-request locks provide the second layer of throttling.
# Only a holder without a dispatched demand may then run the established
# hostile/rival/low-opinion automatic removal rule.
gdd_quarterly_tianxia_territory_maintenance_effect = {
    event_target:EmperorOfChina = {
        clr_country_flag = gdd_tianxia_unlawful_ai_demand_sent_this_tick
    }
    every_country = {
        if = {
            limit = {
                exists = yes
                event_target:EmperorOfChina = {
                    ai = yes
                    NOT = { has_country_flag = gdd_tianxia_unlawful_ai_demand_sent_this_tick }
                }
                gdd_can_receive_tianxia_unlawful_demand_trigger = {
                    emperor = event_target:EmperorOfChina
                }
            }
            set_country_flag = gdd_tianxia_unlawful_demand_considered_this_tick
            save_event_target_as = gdd_tianxia_unlawful_selected_country
            event_target:EmperorOfChina = {
                set_country_flag = gdd_tianxia_unlawful_ai_demand_sent_this_tick
                save_event_target_as = gdd_tianxia_unlawful_emperor
                country_event = { id = gdd_tianxia_territory.10 }
            }
        }
        if = {
            limit = {
                exists = yes
                ai = yes
                zhx_is_tianxia_polity = yes
                is_at_war = no
            }
            every_owned_province = {
                limit = {
                    gdd_can_add_province_to_tianxia_for_country_trigger = { who = PREV }
                }
                gdd_add_province_to_tianxia_effect = yes
            }
        }
        else_if = {
            limit = {
                exists = yes
                ai = yes
                NOT = { zhx_is_tianxia_polity = yes }
                NOT = { has_country_flag = gdd_tianxia_unlawful_request_pending }
                NOT = { has_country_flag = gdd_tianxia_unlawful_demand_considered_this_tick }
                is_at_war = no
                event_target:EmperorOfChina = { exists = yes }
                OR = {
                    is_rival = event_target:EmperorOfChina
                    ai_attitude = {
                        who = event_target:EmperorOfChina
                        attitude = attitude_hostile
                    }
                    NOT = {
                        has_opinion = {
                            who = event_target:EmperorOfChina
                            value = -50
                        }
                    }
                }
            }
            every_owned_province = {
                limit = {
                    gdd_can_remove_province_from_tianxia_for_country_trigger = { who = PREV }
                }
                PREV = { gdd_apply_tianxia_removal_opinion_effect = yes }
                gdd_remove_province_from_tianxia_effect = yes
            }
        }
        every_owned_province = {
            limit = {
                has_province_flag = zhx_tianxia_province
                is_part_of_hre = yes
            }
            set_in_empire = no
        }
        every_owned_province = {
            limit = {
                has_province_flag = zhx_tianxia_province
                is_owned_by_trade_company = yes
            }
            # No scriptable remove-from-trade-company effect exists. Ordinary
            # clicks are blocked by the opaque Zhou/HRE/TC shared-slot overlay;
            # if automation bypasses it, trade-company status wins this safety
            # repair so the province can never retain both memberships.
            clr_province_flag = zhx_tianxia_province
            clr_province_flag = gdd_tianxia_unlawful_request_pending
            clr_province_flag = gdd_tianxia_unlawful_requested_this_owner
            remove_province_modifier = gdd_tianxia_unlawful_refusal
        }
        clr_country_flag = gdd_tianxia_unlawful_demand_considered_this_tick
    }
    event_target:EmperorOfChina = {
        clr_country_flag = gdd_tianxia_unlawful_ai_demand_sent_this_tick
    }
}
