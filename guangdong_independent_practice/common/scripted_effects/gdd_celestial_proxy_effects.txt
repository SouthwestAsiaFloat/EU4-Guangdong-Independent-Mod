# Script-side replacements for the two hard-coded Empire of China actions.
# Decrees keep their existing ownership and costs. Reform transactions use
# ROOT as executor and spend the actor's side of the Zhou authority balance.

gdd_enact_proxy_decree_effect = {
    # Recheck at execution, including events opened before a government change.
    if = {
        limit = { NOT = { has_reform = zhx_tianzi_reform } meritocracy = 20 }
        add_meritocracy = -20
        add_country_modifier = {
            name = $which$
            duration = 3650
        }
    }
}

gdd_enact_proxy_celestial_reform_effect = {
    if = {
        limit = { gdd_proxy_celestial_reform_base_trigger = yes $which$_can_enact = yes NOT = { has_global_flag = $which$ } }

        gdd_pay_reform_authority_cost = yes
        hidden_effect = {
            set_global_flag = $which$
            set_country_flag = $which$
            $which$_on_pass = yes
            CZH = {
                gdd_clear_passed_reform_support_effect = yes
                gdd_recalculate_all_reform_votes_effect = yes
            }
        }
    }
}

# Apply the public-opinion backlash before the executor flag is cleared.
# ROOT is always the country responsible for the revocation. Active revocation
# therefore calls this from the acting Emperor/principal feudatory; a future
# peace term must call it from the victorious country proposing that term.
gdd_apply_proxy_reform_revoke_backlash_effect = {
    every_country = {
        limit = {
            exists = yes
            NOT = { tag = ROOT }
            OR = {
                has_country_flag = $which$
                zhx_is_tianxia_polity = yes
            }
        }
        if = {
            limit = { has_country_flag = $which$ }
            # Five applications use the same capped modifier: -50 for the
            # original executor, while preserving the shared -100 floor.
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
        }
        else_if = {
            limit = { zhx_is_tianxia_polity = yes }
            add_opinion = { who = ROOT modifier = gdd_opinion_revoked_tianxia_reform }
        }
    }
}

# Shared low-level revoke interface. The caller must leave the culpable country
# in ROOT and pass the exact reform token; this keeps the future peace path and
# active path on the same responsibility and backlash rules.
gdd_revoke_proxy_celestial_reform_effect = {
    hidden_effect = {
        gdd_apply_proxy_reform_revoke_backlash_effect = { which = $which$ }
        clr_global_flag = $which$
        every_country = {
            clr_country_flag = $which$
        }
        CZH = {
            gdd_clear_passed_reform_support_effect = yes
            gdd_recalculate_all_reform_votes_effect = yes
        }
    }
}

# Centralisation final: ROOT becomes independent first, then every other
# existing Zhou polity is detached from its current overlord and made ROOT's
# special no-diplomatic-slot vassal.
gdd_unify_tianxia_under_reform_executor_effect = {
    hidden_effect = {
        if = {
            limit = { is_subject = yes }
            grant_independence = yes
        }
        every_country = {
            limit = {
                exists = yes
                zhx_is_tianxia_polity = yes
                NOT = { tag = ROOT }
            }
            if = {
                limit = { is_subject = yes }
                grant_independence = yes
            }
            ROOT = {
                create_subject = {
                    subject_type = gdd_tianxia_vassal
                    who = PREV
                }
            }
        }
        set_global_flag = eoc_vassalize_tributaries
    }
}

# Reserved hook for the future decentralised realm-war system.
gdd_enable_decentralized_tianxia_interface_effect = {
    hidden_effect = {
        set_global_flag = gdd_decentralized_tianxia_interface_enabled
    }
}

gdd_disable_centralized_tianxia_interface_effect = {
    hidden_effect = {
        every_country = {
            limit = {
                exists = yes
                is_subject_of_type = gdd_tianxia_vassal
            }
            grant_independence = yes
        }
        # The native reform may independently own this shared vanilla capability.
        if = {
            limit = { NOT = { empire_of_china_reform_passed = vassalize_tributaries_decision } }
            clr_global_flag = eoc_vassalize_tributaries
        }
    }
}

gdd_disable_decentralized_tianxia_interface_effect = {
    hidden_effect = {
        clr_global_flag = gdd_decentralized_tianxia_interface_enabled
    }
}

# Public wrappers for the two final reforms include their world-state cleanup.
# Future peace terms should use these wrappers rather than the low-level effect.
gdd_revoke_proxy_celestial_final_centralization_effect = {
    gdd_revoke_proxy_celestial_reform_effect = { which = gdd_proxy_reform_vassalize_tributaries }
    gdd_disable_centralized_tianxia_interface_effect = yes
}

gdd_revoke_proxy_celestial_final_decentralization_effect = {
    gdd_revoke_proxy_celestial_reform_effect = { which = gdd_proxy_reform_new_world }
    gdd_disable_decentralized_tianxia_interface_effect = yes
}


# ROOT is the authorized actor. Payment stays on CZH, the existing balance
# anchor, and never moves through zero. Shared by enactment and active repeal.
gdd_pay_reform_authority_cost = {
    if = {
        limit = { gdd_proxy_celestial_reform_base_trigger = yes }
        CZH = {
            if = {
                limit = { check_variable = { which = gdd_central_authority_balance value = 60 } }
                change_variable = { which = gdd_central_authority_balance value = -60 }
            }
            else = {
                change_variable = { which = gdd_central_authority_balance value = 60 }
            }
            gdd_clamp_central_authority_balance = yes
            gdd_refresh_authority_balance_effects = yes
        }
    }
}

# Deliberate paid repeal. The lower-level clear effect remains available for
# system teardown; it cannot be reached by an unpaid player/AI action.
gdd_actively_revoke_proxy_celestial_reform_effect = {
    if = {
        limit = {
            gdd_can_actively_revoke_proxy_celestial_reform_trigger = yes
            has_global_flag = $which$
        }
        gdd_pay_reform_authority_cost = yes
        gdd_revoke_proxy_celestial_reform_effect = { which = $which$ }
        $which$_on_revoke = yes
    }
}
