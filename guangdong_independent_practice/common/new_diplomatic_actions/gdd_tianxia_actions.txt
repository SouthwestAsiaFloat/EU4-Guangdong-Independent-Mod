# Human-facing native diplomacy for the Zhou Tianxia. The engine currently
# never lets AI actors send custom diplomatic actions, so AI Emperors retain
# the quarterly event-backed path in gdd_tianxia_territory_events.txt.

gdd_tianxia_demand_unlawful_territory = {
    # Share the native category localised as Zhou Tianxia with relief and Qinwang.
    category = is_emperor_of_china

    # Reuse the vanilla unlawful-territory message icon. This is a messageicons
    # frame and does not reserve or shift any topbar alert slot.
    alert_index = 22
    alert_tooltip = gdd_tianxia_demand_unlawful_territory_alert_tooltip

    require_acceptance = no

    is_visible = {
        ai = no
        zhx_is_tianzi = yes
        has_global_flag = zhx_system_initialised_v13
        NOT = { has_global_flag = zhx_tianxia_dismantled }
        FROM = {
            exists = yes
            NOT = { tag = ROOT }
        }
    }

    is_allowed = {
        custom_trigger_tooltip = {
            tooltip = gdd_tianxia_picker_already_open_tt
            NOT = { has_country_flag = gdd_tianxia_picker_open }
        }
        custom_trigger_tooltip = {
            tooltip = gdd_tianxia_demand_unlawful_territory_sovereign
            FROM = {
                OR = {
                    is_subject = no
                    is_subject_of = ROOT
                }
            }
        }
        custom_trigger_tooltip = {
            tooltip = gdd_tianxia_demand_unlawful_territory_not_at_war
            NOT = { war_with = FROM }
        }
        custom_trigger_tooltip = {
            tooltip = gdd_tianxia_demand_unlawful_territory_allowed
            FROM = {
                gdd_can_be_tianxia_unlawful_addressee_trigger = {
                    emperor = ROOT
                }
            }
        }
    }

    # This action opens the sender's picker. The actual demand is sent only
    # after a province is selected and confirmed in event .21.
    on_accept = {
        gdd_open_tianxia_unlawful_picker_effect = yes
    }
    ai_will_do = { always = no }
}
