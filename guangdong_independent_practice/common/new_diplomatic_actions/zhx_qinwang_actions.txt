        }
        if = {
            limit = { NOT = { FROM = { owes_favors = { who = ROOT value = 25 } } } }
            custom_trigger_tooltip = { tooltip = zhx_qn_favors_tt FROM = { owes_favors = { who = ROOT value = 25 } } }
        }
        if = {
            limit = { NOT = { FROM = { NOT = { has_country_flag = zhx_qn_sent } NOT = { has_country_flag = zhx_qn_pending } NOT = { has_country_flag = zhx_qn_joining } } } }

            limit = { NOT = { OR = { NOT = { has_global_flag = zhx_qn_active } zhx_qn_original_war = yes } } }
            custom_trigger_tooltip = { tooltip = zhx_qn_campaign_tt OR = { NOT = { has_global_flag = zhx_qn_active } zhx_qn_original_war = yes } }
        }

            limit = { NOT = { FROM = { OR = { ai = no zhx_qn_ai_positive = yes } } } }
            custom_trigger_tooltip = { tooltip = zhx_qn_positive_tt FROM = { OR = { ai = no zhx_qn_ai_positive = yes } } }
        }
    }
    on_accept = {
        custom_tooltip = zhx_qn_sent_tt
        hidden_effect = { if = { limit = { zhx_is_tianzi = yes } FROM = { zhx_qn_dispatch = yes } } }
    }
    on_decline = { }
    ai_will_do = { always = no }
}
