gdd_recover_china_cores_wargoal = {
    type = take_core

    attacker = {
        # Only the attacker's cores inside the China superregion receive the
        # special reconquest benefits described for this CB.
        badboy_factor = 0
        prestige_factor = 1
        peace_cost_factor = 0.10
        peace_cost_factor_non_allowed_treaty = 1.0

        allowed_provinces = {
            is_core = ROOT
            superregion = china_superregion
        }
        allowed_provinces_are_eligible = yes

        peace_options = {
            po_demand_provinces
            po_gold
            po_annul_treaties
        }

        deny_annex = yes
        prov_desc = GDD_RECOVER_CHINA_CORES
    }

    defender = {
        badboy_factor = 1
        prestige_factor = 1
        peace_cost_factor = 1

        peace_options = {
            po_demand_provinces
        }
    }

    war_name = GDD_PUNISH_GUANGDONG_WAR_NAME
}
