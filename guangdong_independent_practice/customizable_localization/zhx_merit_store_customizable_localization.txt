# The merit-store status row is backed by one always-present scripted textbox
# per column. Conditional wording belongs here so inactive alternatives never
# leave an interface object without a custom_gui binding.

defined_text = {
    name = GetZhxMeritStoreRankStatus
    random = no

    text = {
        localisation_key = zhx_merit_store_rank_nonmember
        trigger = { zhx_is_tianxia_member = no }
    }
    text = {
        localisation_key = zhx_merit_store_rank_unqualified
        trigger = {
            zhx_is_tianxia_member = yes
            NOT = { has_country_flag = zhx_merit_active_contribution_this_term }
        }
    }
    text = {
        localisation_key = zhx_merit_store_rank_member
        trigger = {
            zhx_is_tianxia_member = yes
            check_variable = { which = zhx_merit_rank_cache value = 1 }
        }
    }
    text = {
        localisation_key = zhx_merit_store_rank_outside
        trigger = { always = yes }
    }
}
